cmake_minimum_required(VERSION 3.22)
project(rimdroid_slang LANGUAGES C CXX)
if(NOT SLANG_SOURCE_DIR)
    message(FATAL_ERROR "SLANG_SOURCE_DIR must point to the pinned, unmodified Slang source")
endif()
add_subdirectory("${SLANG_SOURCE_DIR}" upstream)
if(ANDROID)
    # Android extracts lib*.so files from an APK, not versioned .so.0.* files.
    # Give the real compiler the name Waylith imports, with the matching SONAME.
    set_target_properties(slang PROPERTIES OUTPUT_NAME slang)
    set_property(TARGET slang PROPERTY VERSION)
    set_property(TARGET slang PROPERTY SOVERSION)
    foreach(module slang-glslang slang-glsl-module)
        target_link_options(${module} PRIVATE "-Wl,-soname,lib${module}-2026.19.so")
    endforeach()
endif()
add_executable(rimdroid-slang-smoke api_smoke.cpp)
set_target_properties(rimdroid-slang-smoke PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/Release/bin")
target_include_directories(rimdroid-slang-smoke PRIVATE "${SLANG_SOURCE_DIR}/include")
target_link_libraries(rimdroid-slang-smoke PRIVATE slang)
target_compile_features(rimdroid-slang-smoke PRIVATE cxx_std_17)
